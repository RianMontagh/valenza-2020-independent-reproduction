cmake_minimum_required(VERSION 3.16)
project(avulsion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(avulsion src/avulsion.cpp)
target_compile_options(avulsion PRIVATE -O3 -Wall -Wextra -Wpedantic)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(avulsion PRIVATE OpenMP::OpenMP_CXX)
endif()
if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(avulsion PRIVATE stdc++fs)
endif()
enable_testing()
add_test(NAME limiting_cases COMMAND avulsion selftest ${CMAKE_BINARY_DIR}/validation.csv)
